1    package br.com.biblioteca.domain;
2    
3    import jakarta.persistence.*;
4    
5    @Entity
6    @Table(
7            name = "fornecedor",
8            uniqueConstraints = @UniqueConstraint(
9                    name = "uk_fornecedor_cnpj",
10                   columnNames = "cnpj"))
11   public class Editora {
12   
13       @Id
14       @GeneratedValue(strategy = GenerationType.IDENTITY)
15       private Long id;
16   
17       @Column(name = "razao_social", nullable = false, length = 150)
18       private String razaoSocial;
19   
20       @Column(nullable = false, length = 14)
21       private String cnpj;
22   
23       @Enumerated(EnumType.STRING)
24       @Column(nullable = false, length = 20)
25       private Status status;
26   
27       protected Editora() {
28       }
29   
30       public Editora(String razaoSocial, String cnpj) {
31           this.razaoSocial = validarTextoObrigatorio(razaoSocial);
32           this.cnpj = validarCnpj(cnpj);
33           this.status = Status.ATIVO;
34       }
35   
36       private String validarTextoObrigatorio(String texto) {
37           if (texto == null || texto.isBlank()) {
38               throw new IllegalArgumentException("Texto obrigatório");
39           }
40   
41           return texto.trim();
42       }
43   
44       private String validarCnpj(String cnpj) {
45           if (cnpj == null || !cnpj.matches("\\d{14}")) {
46               throw new IllegalArgumentException(
47                       "CNPJ deve possuir 14 dígitos");
48           }
49   
50           return cnpj;
51       }
52   
53       public Long getId() {
54           return id;
55       }
56   
57       public String getRazaoSocial() {
58           return razaoSocial;
59       }
60   
61       public String getCnpj() {
62           return cnpj;
63       }
64   
65       public Status getStatus() {
66           return status;
67       }
68   }
